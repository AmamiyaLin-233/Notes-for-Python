#==============================================================================递归 Recursion，动态规划 Dynamic Programming

def f(x):  # 斐波那契数列：f(0) = 0，f(1) = 1，f(x) = f(x-1) + f(x-2)
    if x == 0: return 0
    if x == 1: return 1
    return f(x - 1) + f(x - 2)  # 递归 Recursion

d = {0: 0, 1: 1} #  记忆化递归：自顶向下 Top-Down DP
def f(x):
    if x in d : return d[x]
    d[x] = f(x - 1) + f(x - 2)
    return d[x]

def f(x): # 迭代循环：自底向上 Bottom-Up DP
    d = {0: 0, 1: 1}
    if x in d : return d[x]
    for i in range(2, x+1):
        d[i] = d[i-1] + d[i-2]
    return d[x]

#==============================================================================滑动窗口 Sliding Window

s = 'dskjfaigks'  # 无重复字符的最长子串长度

ans = 1
if len(s) <= 1:
    ans = len(s)
else:
    wind = {s[0]: 0}
    left = 0
    for right in range(1, len(s)):
        if s[right] in wind:
            ans = max(ans, right - left)
            newleft = wind[s[right]] + 1
            for i in range(left, newleft):
                wind.pop(s[i])
            left = newleft
        else:
            ans = max(ans, right - left + 1)
        wind[s[right]] = right
print(ans)

#==============================================================================广度优先搜索 Breadth-First Search (BFS)

nums = [1, 2, 3] # 全排列

def f(nums: list[int]) -> list[list[int]]:
    curr = [[]]
    for l in range(1, len(nums) + 1): # 逐层检索
        nex = []
        for c in curr:  # 逐个提取上一步的结果
            for n in nums:
                if n not in c: # 剪枝
                    nex.append(c + [n])  # 列表拼接，例如：[1,2] + [3] = [1,2,3]
        curr = nex  # 关键！重新指向
    return curr

print(f(nums))

#==============================================================================回溯 Backtracking， 深度优先搜索 Depth First Search

nums = [1, 2, 3]  # 全排列

def f(nums: list[int]) -> list[list[int]]:
    result = []
    path = []  # 维护当前路径
    n = len(nums)
    is_able = [True] * n  # 维护剩余可选项

    def g():  # 回溯递归函数，深度优先搜索
        if len(path) == n:  # 终止条件
            result.append(path.copy())
            return
        else:
            for ind, num in enumerate(nums):  # 获取所有的可选项
                if is_able[ind]:  # 剪枝
                    path.append(num)  # 保存结果
                    is_able[ind] = False

                    g()  # 钻入下一层

                    path.pop()  # 关键！恢复状态
                    is_able[ind] = True

    g()
    return result

print(f(nums))

class ListNode: # 链表

    def __init__(self, val=0, next=None): # 初始方法
        self.val = val
        self.next = next

    @classmethod # 类方法
    def from_list(cls, nodelist): 
        if not nodelist:
            return None

        head = cls(nodelist[0])
        curr = head
        for val in nodelist[1:]:
            curr.next = cls(val)
            curr = curr.next
        return head

class TreeNode:  
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left  
        self.right = right  

    @classmethod
    def from_list(cls, arr):  # 构建用list导入的方法
        if not arr or arr[0] is None:
            return None
        root = cls(arr[0])
        queue = [root]
        i = 1
        while queue and i < len(arr):
            current = queue.pop(0)
            if i < len(arr) and arr[i] is not None:
                current.left = cls(arr[i])  
                queue.append(current.left)
            i += 1
            if i < len(arr) and arr[i] is not None:
                current.right = cls(arr[i])  
                queue.append(current.right)
            i += 1
        return root


arr = [3, 9, 20, None, None, 15, 7]
root = TreeNode.from_list(arr)


def print_level_order(root):  # 打印整棵树为 list
    if not root:
        print("None")
        return
    queue = [root]
    result = []
    while queue:
        node = queue.pop(0)
        if node:
            result.append(str(node.val))
            queue.append(node.left)
            queue.append(node.right)
        else:
            result.append("None")
    while result and result[-1] == "None":
        result.pop()
    print(result)


print_level_order(root)
